models/index.js

const {DataTypes } =require("sequelize");
const sequelize = require("../db");
const db ={};

db.Product = require("./product")(sequelize,DataTypes);
db.Category= require("./category")(sequelize,DataTypes);

db.Category.hasMany(db.Product,{
   foreignKey:"category_id",
});

db.Product.belongsTo(db.Category,{

   foreignKey:"category_id",
});


db.sequelize = sequelize;
module.exports = db;

Types of Joins in Sequelize

Sequelize me mainly 3 joins use hote hain:

Join Type	Sequelize Option	Meaning
INNER JOIN	required: true	Sirf matching data
LEFT JOIN	default (required: false)	Sab left + matching right
RIGHT JOIN	right: true	Sab right + matching left
