--Create Table
--Create a table students with:
--id (auto increment, primary key)
--name (not null)
--email (unique)
--age (must be ≥ 18)


create table students(
id serial primary key,
name varchar(50) not null,
email varchar(50) unique,
age int check(age >= 18)
);

--Q2:Get:user name,order id,product id (users + orders + order_items)
select u.name,o.ord_id,oi.prod_id from practice.orders o 
left JOIN practice.users u on u.user_id = o.user_id
left JOIN practice.order_items oi on o.ord_id = oi.ord_id;


SELECT u.name, o.ord_id AS order_id, oi.prod_id
FROM practice.users u
JOIN practice.orders o 
  ON u.user_id = o.user_id
JOIN practice.order_items oi 
  ON o.ord_id  = oi.ord_id;


--Q3:Get:--user name--product name--quantity
--(users + orders + order_items + products)


select u.name,p.name,oi.quantity from practice.users as u
JOIN practice.orders as o on u.user_id = o.user_id
Join practice.order_items as oi on o.ord_id = oi.ord_id
JOIN practice.products as p on oi.prod_id = p.prod_id;




-- Q4:Get:user name,product name,total price (price * quantity)

select u.name,p.name,(p.price*oi.quantity) as total_price from practice.users as u
